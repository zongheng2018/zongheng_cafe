    async function addMenuItem() {

      const price = parseInt(document.getElementById("price").value);

        item: item,

      alert("品項已新增！");

      const snapshot = await db.collection("menu").get();

        const menuItem = doc.data();

        li.innerText = menuItem.item + " - " + menuItem.price + " 元";

        list.appendChild(li);

      });
